Startup Instructions 
# 1. Database: Start PostgreSQL & ensure connection string in appsettings.json
# 2. Agentic AI & Price API:
cd ai_agent && pip install -r requirements.txt && python -m uvicorn main:app --port 8000
cd price_api && python main.py

# 3. ASP.NET Core Web API:
cd FishLink.API && dotnet run   # Runs on http://localhost:5157

# 4. React Web Dashboard:
cd fishlink-dashboard && npm start   # Runs on http://localhost:3000

# 5. Flutter Mobile Application:
cd fishlink_mobile && flutter run -d chrome --web-port=8080   # Web demo
# (Or release APK): flutter build apk --release
